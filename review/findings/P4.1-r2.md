```bash
import runpy
m = runpy.run_path('tests/test_triage_gate.py')
for source in ('rule', 'agent'):
    base = {'fraud_signal': {
        'status': 'pass', 'scope_match': True, 'window_complete': True,
    }}
    for status in ('absent', 'fail', 'not_applicable'):
        results = dict(base)
        if status != 'absent':
            results['seasonal_calendar'] = {
                'status': status, 'scope_match': False,
            }
        row = m['_gate'](
            m['_proposal']('fraud_ops', cause='fraud_surge', runbook=None,
                           checklist_results=results),
            source=source,
        )
        print(source, status, '=>', row['action'], row['evidence_status'])
PY
```

```text
absent => fraud_ops sufficient
```
